Add this item near the top of the existing AccountantDashboardClient.tsx navigation array:

{ page: "Operations Control Center", glyph: "dashboard", group: "Overview" },

Because the current PageKey implementation renders internal sections, handle this item before setActivePage:

function openPage(page: PageKey | "Operations Control Center") {
  if (page === "Operations Control Center") {
    router.push("/accountant/control-center");
    return;
  }

  setActivePage(page);
  setMobileOpen(false);
  setNoticeOpen(false);
}
